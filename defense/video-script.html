    [McpServerTool(Name = "search_games", ReadOnly = true, OpenWorld = false), Description("Search Gamertab's saved game catalogue by title. Returns local game IDs for add_to_library. This does not search the live IGDB catalogue.")]
    public async Task<string> SearchGames([Description("Part of the game title, up to 120 characters.")] string query = "", [Description("Maximum results, from 1 to 20.")] int limit = 10)
    {
        if (query is null || query.Length > 120 || limit is < 1 or > 20) throw new McpException("Use a title up to 120 characters and a limit from 1 to 20.");
        await using var database = await databases.CreateDbContextAsync();
        var term = query.Trim().ToLowerInvariant();
        return Json(await database.Games.AsNoTracking().Where(game => game.Name.ToLower().Contains(term)).OrderBy(game => game.Name)
            .Take(limit).Select(game => new { game.Id, game.Name, game.Genres, game.ReleasedOn }).ToListAsync());
    }

    [McpServerTool(Name = "get_library", ReadOnly = true, OpenWorld = false), Description("Read the configured account's private game library, statuses and personal ratings. No other account can be selected.")]
    public async Task<string> GetLibrary([Description("Items to skip, from 0 to 100000.")] int offset = 0, [Description("Page size, from 1 to 100.")] int limit = 20)
    {
        if (offset is < 0 or > 100000 || limit is < 1 or > 100) throw new McpException("Use an offset from 0 to 100000 and a limit from 1 to 100.");
        await using var database = await databases.CreateDbContextAsync();
        return Json(await database.LibraryEntries.AsNoTracking().Where(entry => entry.UserId == account.UserId).OrderBy(entry => entry.Id)
            .Skip(offset).Take(limit).Select(entry => new { entry.Id, entry.GameId, gameName = entry.Game.Name, status = entry.Status.ToString(), entry.PersonalRating }).ToListAsync());
    }

    [McpServerTool(Name = "add_to_library", ReadOnly = false, Destructive = false, Idempotent = true, OpenWorld = false), Description("Add a saved catalogue game to the configured account's library as Planned. Only call when the user asks to add it. Existing entries, statuses, notes and ratings are preserved.")]
    public async Task<string> AddToLibrary([Description("Local game ID returned by search_games, not a Steam or IGDB ID.")] int gameId)
    {
        await using var database = await databases.CreateDbContextAsync();
        var game = await database.Games.AsNoTracking().FirstOrDefaultAsync(game => game.Id == gameId);
        if (game is null) throw new McpException("Game not found. Search the saved catalogue first.");
        var added = await database.Database.ExecuteSqlInterpolatedAsync($"""
            INSERT INTO "LibraryEntries" ("UserId", "GameId", "Status", "Notes", "FollowNews")
            VALUES ({account.UserId}, {gameId}, 0, {""}, {true}) ON CONFLICT ("UserId", "GameId") DO NOTHING
            """) == 1;
        var entry = await database.LibraryEntries.AsNoTracking().SingleAsync(entry => entry.UserId == account.UserId && entry.GameId == gameId);
        return Json(new { added, entry.Id, entry.GameId, gameName = game.Name, status = entry.Status.ToString() });

Короткі відповіді на запитання

Чому сайт не Netlify

Це ASP.NET сервер із PostgreSQL, приватними файлами й живими з’єднаннями. Cloud Run запускає наш Docker-контейнер, а не тільки статичні файли.

Де доказ B7, якщо експорт є локально

На публічному сайті LibraryExportClient надсилає запит до окремого приватного сервісу через IAM token. Видно окремий image, deployment і запит у Cloud Run. Локально викликається formatter у тому самому процесі.

Як довести B9

OutputCacheTests змінює БД після першого GET і перевіряє незмінну кешовану відповідь. Інший тест перевіряє скидання кешу після імпорту. Мала затримка HTTP сама по собі не доказ.

Чи C13/C14 дають доступ іншим користувачам

Ні. Це дві відкриті копії одного приватного документа одного акаунта. Група SignalR і перевірка власника прив’язані до user ID та гри.

Що відбувається при одночасному збереженні

UPDATE має умову за Revision. Лише одна зміна приймається, інша отримує 409. UI зберігає чернетку й пропонує вибрати версію. Це контроль конфліктів, не CRDT.

Чому 84,61 відсотка і 28,24 відсотка одночасно

28,24 охоплює всі C# рядки разом з великими згенерованими міграціями. 84,61 є перерахунком того самого Cobertura без Data/Migrations. Браузерні результати наведені окремо. JavaScript не вимірювався.

Чи всі тести ходять у реальні API

Ні. Повторювані інтеграційні й браузерні тести підміняють зовнішні HTTP-відповіді. Окремий живий аудит перевірив IGDB, Gemini, HubSpot, MCP, GCS та приватну функцію.

Чи Google-вхід обмежений одним тестовим акаунтом

Запит на локальному й публічному сайтах містить лише openid, profile та email. Для цих базових scopes Google документує виняток із тестового allowlist. Режим Testing залишений, production branding не заявляється. Корпоративний адміністратор усе одно може обмежити сторонній вхід.

Чи Gemini й MCP це одна функція

Ні. Gemini є вебчатом із трьома read-only інструментами. MCP є локальним stdio-сервером для зовнішнього агента і має також один явний інструмент додавання гри.

Чому немає Metacritic для кожної гри

Steam повертає доступний Metacritic для PC. Не кожна гра має цей показник і Steam-сторінку. Відсутнє значення приховується. Ціни інших магазинів не заявляються.

Що буде з двома екземплярами Cloud Run

Поточне розгортання обмежене одним екземпляром. Історія та документи в PostgreSQL, але live-сигнали в пам’яті процесу. Для горизонтального масштабування потрібен SignalR backplane або інший спільний механізм сповіщення.

Чи це безкоштовно назавжди

Поточний хостинг використовує активований Google Cloud trial. Ліміти max 1, min 0 і budget alert обмежують витрати, але це не обіцянка вічного безкоштовного хостингу. Не потрібно переходити на платний тариф для захисту.

Чому C8 не враховано

Мапи на сайті немає. Виклики зовнішніх API самі по собі не підміняють вимогу інтеграції мапи, тому цей пункт у суму не доданий.

Офіційне пояснення Google про базові scopes

Як діяти при затримці

Якщо Cloud Run прокидається, дати сторінці завантажитися, поки пояснюєш архітектуру. Якщо зовнішній API повернув квоту чи помилку, показати повідомлення й назвати залежність. Не називати старий запис живою відповіддю. Для тестів відкрити журнал перевіреного запуску та команду повторення. Якщо змінився ID демозапису, знайти гру через My library, а не підставляти старе число.
